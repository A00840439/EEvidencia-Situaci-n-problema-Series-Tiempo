# Function to clean series (handle NaN/Inf)
def clean_series(series, name):
    try:
        print(f"Checking {name} for NaN/Inf values...")
        if not pd.api.types.is_numeric_dtype(series):
            try:
                series = pd.to_numeric(series, errors='coerce')
            except:
                raise ValueError(f"{name} contains non-numeric values that cannot be converted.")
        if series.isna().any() or np.isinf(series).any():
            print(f"{name} contains NaN or Inf values. Cleaning...")
            series = series.fillna(method='ffill').fillna(method='bfill').dropna()
            series = series.replace([np.inf], series.max())
            series = series.replace([-np.inf], series.min())
        if series.isna().any():
            raise ValueError(f"{name} still contains NaN values after cleaning.")
        if len(series) == 0:
            raise ValueError(f"{name} is empty after cleaning.")
        print(f"{name} is clean with {len(series)} observations.")
        return series
    except Exception as e:
        raise ValueError(f"Failed to clean {name}: {str(e)}")

# Function to ensure datetime index
def ensure_datetime_index(df, series, name):
    try:
        if not pd.api.types.is_datetime64_any_dtype(df.index):
            datetime_cols = ['Date', 'Timestamp', 'datetime', 'time']
            for col in datetime_cols:
                if col in df.columns:
                    df[col] = pd.to_datetime(df[col], errors='coerce')
                    if df[col].isna().all():
                        raise ValueError(f"Invalid datetime values in {col} for {name}.")
                    df.set_index(col, inplace=True)
                    series.index = df.index
                    break
            else:
                print(f"Warning: No datetime column found for {name}. Using default index.")
                df.index = pd.RangeIndex(len(df))
                series.index = df.index
        return df, series
    except Exception as e:
        print(f"Warning: Failed to set datetime index for {name}: {str(e)}. Using default index.")
        df.index = pd.RangeIndex(len(df))
        series.index = df.index
        return df, series

# Function to perform stationarity tests
def stationarity_tests(series, name):
    try:
        print(f"\nStationarity Tests for {name}")
        series = clean_series(series, name)

        adf_result = adfuller(series, autolag='AIC')
        print(f"ADF Test p-value: {adf_result[1]:.4f}")
        kpss_result = kpss(series, regression='c', nlags='auto')
        print(f"KPSS Test p-value: {kpss_result[1]:.4f}")

        plt.figure(figsize=(12, 6))
        plt.plot(series, label='Original Series', color='blue')
        plt.title(f'Original {name} Series')
        plt.xlabel('Time')
        plt.ylabel('Price')
        plt.grid(True)
        plt.legend()
        filename = f'{name.replace(" ", "_")}_original_series.png'
        plt.savefig(filename)
        plt.show()
        plt.close()
        print(f"Saved and displayed plot: {filename}")

        is_differenced = False
        diff_series = series
        if adf_result[1] > 0.05 or kpss_result[1] < 0.05:
            print(f"{name} is non-stationary. Differencing...")
            diff_series = series.diff().dropna()
            diff_series = clean_series(diff_series, f"Differenced {name}")
            plt.figure(figsize=(12, 6))
            plt.plot(diff_series, label='Differenced Series', color='orange')
            plt.title(f'Differenced {name} Series')
            plt.xlabel('Time')
            plt.ylabel('Price Difference')
            plt.grid(True)
            plt.legend()
            filename = f'{name.replace(" ", "_")}_differenced_series.png'
            plt.savefig(filename)
            plt.show()
            plt.close()
            print(f"Saved and displayed plot: {filename}")
            is_differenced = True
        return diff_series, is_differenced
    except Exception as e:
        print(f"Warning: Stationarity tests failed for {name}: {str(e)}. Using original series.")
        return series, False

# Function to perform cointegration test
def cointegration_test(series1, series2, name1, name2):
    try:
        print(f"\nCointegration Test between {name1} and {name2}")
        df = pd.DataFrame({name1: series1, name2: series2}).dropna()
        series1, series2 = df[name1], df[name2]
        if len(series1) < 2:
            raise ValueError("Insufficient data for cointegration test.")

        series1 = clean_series(series1, name1)
        series2 = clean_series(series2, name2)

        score, p_value, _ = coint(series1, series2)
        print(f"Engle-Granger Cointegration Test p-value: {p_value:.4f}")
        if p_value < 0.05:
            print("Series are cointegrated")
        else:
            print("Series are not cointegrated")

        plt.figure(figsize=(12, 6))
        plt.plot(series1, label=name1, color='blue')
        plt.plot(series2, label=name2, color='green')
        plt.title(f'{name1} vs {name2} for Cointegration')
        plt.xlabel('Time')
        plt.ylabel('Price')
        plt.grid(True)
        plt.legend()
        filename = f'{name1.replace(" ", "_")}_vs_{name2.replace(" ", "_")}_cointegration.png'
        plt.savefig(filename)
        plt.show()
        plt.close()
        print(f"Saved and displayed plot: {filename}")

        return series1, series2
    except Exception as e:
        print(f"Warning: Cointegration test failed: {str(e)}. Skipping.")
        return series1, series2

# Function to fit and forecast models
def fit_and_forecast(series, name, steps=10):
    try:
        print(f"\nFitting Models for {name}")
        series = clean_series(series.dropna(), name)
        if len(series) < 50:
            print(f"Warning: Insufficient data for {name}: {len(series)} observations. Proceeding with limited data.")

        train_size = int(len(series) * 0.8)
        if train_size < 10:
            raise ValueError(f"Training set too small for {name}: {train_size} observations.")
        train, test = series[:train_size], series[train_size:]
        if len(test) < steps:
            print(f"Warning: Test set too small ({len(test)} < {steps}). Adjusting steps to {len(test)}.")
            steps = len(test)

        ar_forecast = np.full(steps, np.nan)
        try:
            ar_model = AutoReg(train, lags=1).fit()
            ar_forecast = ar_model.forecast(steps=steps)
            print("AR Model Summary:")
            print(ar_model.summary())
        except Exception as e:
            print(f"Warning: AR model failed for {name}: {str(e)}")

        arma_forecast = np.full(steps, np.nan)
        try:
            arma_model = ARIMA(train, order=(1, 0, 1)).fit()
            arma_forecast = arma_model.forecast(steps=steps)
            print("ARMA Model Summary:")
            print(arma_model.summary())
        except Exception as e:
            print(f"Warning: ARMA model failed for {name}: {str(e)}")

        arima_forecast = np.full(steps, np.nan)
        try:
            arima_model = ARIMA(train, order=(1, 1, 1)).fit()
            arima_forecast = arima_model.forecast(steps=steps)
            print("ARIMA Model Summary:")
            print(arima_model.summary())
        except Exception as e:
            print(f"Warning: ARIMA model failed for {name}: {str(e)}")

        return train, test, ar_forecast, arma_forecast, arima_forecast, steps
    except Exception as e:
        print(f"Warning: Forecasting failed for {name}: {str(e)}")
        return None, None, np.full(steps, np.nan), np.full(steps, np.nan), np.full(steps, np.nan), steps

# Function to create combined graph

        if len(nu_series) > 0:
            axes[0, 0].plot(nu_series.index, nu_series, label='Original', color='blue')
            if nu_is_diff and len(nu_diff) > 0:
                axes[0, 0].plot(nu_diff.index, nu_diff, label='Differenced', color='orange', alpha=0.5)
            axes[0, 0].set_title('NU Close: Stationarity')
            axes[0, 0].set_xlabel('Time')
            axes[0, 0].set_ylabel('Price / Diff')
            axes[0, 0].grid(True)
            axes[0, 0].legend()
        else:
            axes[0, 0].text(0.5, 0.5, 'No NU data', ha='center', va='center')

        # Honeywell: Original and Differenced Series
        if len(hon_series) > 0:
            axes[0, 1].plot(hon_series.index, hon_series, label='Original', color='blue')
            if hon_is_diff and len(hon_diff) > 0:
                axes[0, 1].plot(hon_diff.index, hon_diff, label='Differenced', color='orange', alpha=0.5)
            axes[0, 1].set_title('Honeywell Close: Stationarity')
            axes[0, 1].set_xlabel('Time')
            axes[0, 1].set_ylabel('Price / Diff')
            axes[0, 1].grid(True)
            axes[0, 1].legend()
        else:
            axes[0, 1].text(0.5, 0.5, 'No Honeywell data', ha='center', va='center')

            axes[1, 0].set_xlabel('Time')
            axes[1, 0].set_ylabel('Normalized Price')
            axes[1, 0].grid(True)
            axes[1, 0].legend()
        else:
            axes[1, 0].text(0.5, 0.5, 'No data for cointegration', ha='center', va='center')

        # Empty subplot
        axes[1, 1].axis('off')

        # NU: Forecasts
        if nu_train is not None and len(nu_train) > 0:
            train_plot_size = min(50, len(nu_train))

            axes[2, 0].plot(nu_test.index[:nu_steps], nu_test[:nu_steps], label='Test', color='black')
            if not np.isnan(nu_ar).all():
                axes[2, 0].plot(nu_test.index[:nu_steps], nu_ar, label='AR', color='red')
            if not np.isnan(nu_arma).all():
                axes[2, 0].plot(nu_test.index[:nu_steps], nu_arma, label='ARMA', color='green')
            if not np.isnan(nu_arima).all():
                axes[2, 0].plot(nu_test.index[:nu_steps], nu_arima, label='ARIMA', color='purple')
            rolling_mean = nu_train.rolling(window=10).mean()
            axes[2, 0].plot(nu_train.index[-train_plot_size:], rolling_mean[-train_plot_size:], label='10-Period Mean', color='orange', linestyle='--')
            axes[2, 0].set_title('NU Close: Forecasts')
            axes[2, 0].set_xlabel('Time')
            axes[2, 0].set_ylabel('Price')
            axes[2, 0].grid(True)
            axes[2, 0].legend()
        else:
            axes[2, 0].text(0.5, 0.5, 'No NU forecast data', ha='center', va='center')

        # Honeywell: Forecasts
        if hon_train is not None and len(hon_train) > 0:
            train_plot_size = min(50, len(hon_train))
            axes[2, 1].plot(hon_train.index[-train_plot_size:], hon_train[-train_plot_size:], label='Train', color='blue')
            axes[2, 1].plot(hon_test.index[:hon_steps], hon_test[:hon_steps], label='Test', color='black')
            if not np.isnan(hon_ar).all():
                axes[2, 1].plot(hon_test.index[:hon_steps], hon_ar, label='AR', color='red')
            if not np.isnan(hon_arma).all():
                axes[2, 1].plot(hon_test.index[:hon_steps], hon_arma, label='ARMA', color='green')
            if not np.isnan(hon_arima).all():
                axes[2, 1].plot(hon_test.index[:hon_steps], hon_arima, label='ARIMA', color='purple')
            rolling_mean = hon_train.rolling(window=10).mean()
            axes[2, 1].plot(hon_train.index[-train_plot_size:], rolling_mean[-train_plot_size:], label='10-Period Mean', color='orange', linestyle='--')
            axes[2, 1].set_title('Honeywell Close: Forecasts')

# Function to plot individual forecasts
def plot_individual_forecasts(train, test, ar_forecast, arma_forecast, arima_forecast, name, steps=10):
    try:
        models = {'AR': ar_forecast, 'ARMA': arma_forecast, 'ARIMA': arima_forecast}
        for model_name, forecast in models.items():
            if len(forecast) != steps or np.isnan(forecast).all():
                print(f"Warning: Invalid {model_name} forecast for {name}. Skipping plot.")
                continue
            plt.figure(figsize=(12, 6))
            train_plot_size = min(50, len(train))
            plt.plot(train.index[-train_plot_size:], train[-train_plot_size:], label='Train', color='blue')
            plt.plot(test.index[:steps], test[:steps], label='Test', color='black')
            plt.plot(test.index[:steps], forecast, label=f'{model_name} Forecast', color='red')
            plt.title(f'{model_name} Forecasting for {name}')
            plt.xlabel('Time')
            plt.ylabel('Price')
            plt.grid(True)
            plt.legend()
            filename = f'{name.replace(" ", "_")}_{model_name}_forecast.png'
            plt.savefig(filename)
            plt.show()
            plt.close()
            print(f"Saved and displayed plot: {filename}")
            display.display(display.Image(filename))
    except Exception as e:
        print(f"Warning: Failed to plot individual forecasts for {name}: {str(e)}")

# Main execution
def main():
    try:
        # Load data
        nu_df, hon_df = load_stock_data()
        nu_close = nu_df['Close']
        hon_close = hon_df['Close']

        # Ensure datetime indices
        nu_df, nu_close = ensure_datetime_index(nu_df, nu_close, "NU Close")
        hon_df, hon_close = ensure_datetime_index(hon_df, hon_close, "Honeywell Close")

        # Stationarity tests
        nu_series, nu_is_diff = stationarity_tests(nu_close, "NU Close")
        hon_series, hon_is_diff = stationarity_tests(hon_close, "Honeywell Close")

        # Cointegration test
        nu_series, hon_series = cointegration_test(nu_series, hon_series, "NU Close", "Honeywell Close")

        # Fit models and forecast
        nu_train, nu_test, nu_ar, nu_arma, nu_arima, nu_steps = fit_and_forecast(nu_series, "NU Close")
        hon_train, hon_test, hon_ar, hon_arma, hon_arima, hon_steps = fit_and_forecast(hon_series, "Honeywell Close")

        # Individual forecast plots
        if nu_train is not None and len(nu_train) > 0:
            plot_individual_forecasts(nu_train, nu_test, nu_ar, nu_arma, nu_arima, "NU Close", nu_steps)
        else:
            print("Warning: No valid training data for NU forecasts. Skipping individual forecast plots.")
        if hon_train is not None and len(hon_train) > 0:
            plot_individual_forecasts(hon_train, hon_test, hon_ar, hon_arma, hon_arima, "Honeywell Close", hon_steps)
        else:
            print("Warning: No valid training data for Honeywell forecasts. Skipping individual forecast plots.")

        # Combined graph
        if len(nu_series) > 0 or len(hon_series) > 0:
            plot_combined_graph(
                nu_close, nu_series if nu_is_diff else nu_close, nu_is_diff,
                hon_close, hon_series if hon_is_diff else hon_close, hon_is_diff,
                nu_train, nu_test, nu_ar, nu_arma, nu_arima, nu_steps,
                hon_train, hon_test, hon_ar, hon_arma, hon_arima, hon_steps
            )
        else:
            print("Warning: No valid data for combined graph.")

        # List and download saved plots
        print("\nSaved Plots:")
        saved_files = [f for f in os.listdir() if f.endswith('.png')]
        if saved_files:
            for f in saved_files:
                print(f"- {f}")
                try:
                    files.download(f)
                except Exception as e:
                    print(f"Warning: Failed to download {f}: {str(e)}")
        else:
            print("No plots were saved.")
    except Exception as e:
        print(f"Error in main execution: {str(e)}")

if __name__ == "__main__":
    main()
